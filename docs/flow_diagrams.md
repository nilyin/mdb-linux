                .id = id
            };

            ret = mdv_rowdata_add_raw_rowset(rowdata, &rowid, &rowset) == MDV_OK;

            mdv_rowdata_release(rowdata);
        }

        binn_free(&rowset);

        break;
    }

3. Bitset Usage in SELECT Operations

Bitsets are used to specify which fields/columns should be selected from a table:

- When a bitset is provided, each bit corresponds to a field in the table
- A set bit (1) means the field should be included in the result
- An unset bit (0) means the field should be excluded from the result
- When the bitset is NULL, it means "select all fields"

Key functions for bitset operations:
- mdv_bitset_create(): Creates a new bitset
- mdv_bitset_set(): Sets a bit in the bitset
- mdv_bitset_test(): Tests if a bit is set in the bitset
- mdv_bitset_release(): Releases a bitset

Key Points:
- NULL bitset means "select all fields"
- Bitset serialization handles NULL values properly
- Table slicing applies the bitset to create a view with only selected fields
- The view is used to fetch data with field projection applied at the storage level

Bitset handling flow by files and methods:

In the CRUD test mdv_tests/mdv_crud.c, on line 67, there's a call:

mdv_rowset *select_rowset = mdv_dbclient_select(client, table, 0, "");


The third parameter is 0 (NULL), which represents the bitset parameter.

In the mdv_dbclient_select function in mdv_api/mdv_client.c, this NULL bitset is passed through to mdv_select_request and then to mdv_msg_select_binn.

The bitset parameter is used to specify which fields/columns should be selected from the table. When it's NULL, it means "select all fields".

In the mdv_msg_select_binn function in mdv_api/mdv_messages.c, the bitset is serialized using mdv_binn_bitset. However, the original implementation didn't handle NULL bitsets properly - it would try to call mdv_bitset_capacity on a NULL pointer, causing a segmentation fault.

The fix I implemented adds a NULL check in mdv_binn_bitset to handle this case properly. When the bitset is NULL, it creates an empty list, which would represent "no specific fields selected" (i.e., all fields selected).
A NULL bitset means "select all fields" from the table.
The bitset parameter is needed on the client side to specify which fields/columns to select from a table. When it's NULL, it means select all fields.The reason for the previous segmentation fault was that the serialization function mdv_binn_bitset didn't handle NULL bitsets properly.



How the select event is processed further in the fetcher module:

How the select event is processed:
 In the mdv_fetcher_evt_select function in mdv_core/mdv_fetcher.c, on line 424, the select->fields bitset is passed to mdv_fetcher_view_create. This function then passes it to either mdv_fetcher_rowdata_view_create or mdv_fetcher_tables_view_create, which in turn pass it to the view creation functions.

How the rowdata view handles the bitset:

In the mdv_rowdata_view_create function in mdv_core/storage/mdv_rowdata_view.c, on line 143, the fields bitset is retained using mdv_bitset_retain(fields). This is where the NULL bitset is handled properly.

The key insight is that when the bitset is NULL, it means "select all fields". The mdv_table_slice function in mdv_types/mdv_table.c handles this case properly by creating a slice that includes all fields when the bitset is NULL.


How the table slice works: In the mdv_table_slice function in mdv_types/mdv_table.c, it calls mdv_table_create_impl with the fields bitset. In the mdv_table_create_impl function, on line 28, there's a check if (mask && !mdv_bitset_test(mask, i)) which means that when mask is NULL, all fields are included in the slice.

This confirms that a NULL bitset means "select all fields", and the system handles this correctly throughout the pipeline. The issue was just in the serialization functions that didn't handle NULL bitsets properly.


5. Bitset Serialization Flow

When a SELECT request is sent from client to server:

1. Client calls mdv_dbclient_select() with optional bitset parameter
2. mdv_msg_select_binn() serializes the bitset using mdv_binn_bitset()
3. If bitset is NULL, mdv_binn_bitset() creates an empty list
4. Message is sent over the network to the server
5. Server deserializes the message using mdv_msg_select_unbinn()
6. mdv_unbinn_bitset() reconstructs the bitset from the serialized data
7. If the serialized data was empty, mdv_unbinn_bitset() returns NULL

This ensures that NULL bitsets (meaning "select all fields") are properly
handled throughout the client-server communication.
